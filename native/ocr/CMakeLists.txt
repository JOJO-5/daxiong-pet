cmake_minimum_required(VERSION 3.18)
project(daxiong_ocr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs)
if(NOT OpenCV_FOUND)
  find_path(OpenCV_INCLUDE_DIRS opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
  foreach(component core imgproc imgcodecs)
    find_library(OpenCV_${component} opencv_${component} REQUIRED)
    list(APPEND OpenCV_LIBS ${OpenCV_${component}})
  endforeach()
endif()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ORT_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${ORT_ROOT}/lib" REQUIRED)
add_executable(daxiong-ocr main.cpp vendor/clipper.cpp)
target_include_directories(daxiong-ocr PRIVATE ${ORT_INCLUDE} ${OpenCV_INCLUDE_DIRS} vendor)
target_link_libraries(daxiong-ocr PRIVATE ${ORT_LIBRARY} ${OpenCV_LIBS})
set_target_properties(daxiong-ocr PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib" INSTALL_RPATH "$ORIGIN")
if(APPLE)
  set_target_properties(daxiong-ocr PROPERTIES INSTALL_RPATH "@loader_path")
endif()
install(TARGETS daxiong-ocr RUNTIME DESTINATION .)
if(WIN32)
  # OpenCV static packages use /MT. Match the worker instead of mixing /MD.
  set_property(TARGET daxiong-ocr PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()
